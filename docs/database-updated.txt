```sql
-- Materialized view para performance
CREATE MATERIALIZED VIEW mv_course_stats AS
SELECT 
  c.id as course_id,
  c.title,
  COUNT(DISTINCT e.user_id) as total_enrollments,
  AVG(e.progress_percentage) as avg_progress,
  COUNT(DISTINCT CASE WHEN e.completed_at IS NOT NULL THEN e.user_id END) as completions,
  (COUNT(DISTINCT CASE WHEN e.completed_at IS NOT NULL THEN e.user_id END)::FLOAT / 
   NULLIF(COUNT(DISTINCT e.user_id), 0) * 100) as completion_rate,
  AVG(EXTRACT(EPOCH FROM (e.completed_at - e.enrolled_at))/3600) as avg_completion_hours
FROM courses c
LEFT JOIN enrollments e ON c.id = e.course_id
WHERE c.status = 'published'
GROUP BY c.id, c.title;

CREATE UNIQUE INDEX ON mv_course_stats (course_id);
```

### Auto-calculate enrollment progress

```sql
CREATE OR REPLACE FUNCTION calculate_enrollment_progress()
RETURNS TRIGGER AS $$
DECLARE
  total_lessons INTEGER;
  completed_lessons INTEGER;
  new_progress DECIMAL(5,2);
BEGIN
  -- Contar total de lições do curso
  SELECT COUNT(*)
  INTO total_lessons
  FROM lessons l
  JOIN modules m ON l.module_id = m.id
  JOIN enrollments e ON m.course_id = e.course_id
  WHERE e.id = NEW.enrollment_id;
  
  -- Contar lições completas
  SELECT COUNT(*)
  INTO completed_lessons
  FROM lesson_progress lp
  WHERE lp.enrollment_id = NEW.enrollment_id
    AND lp.status = 'completed';
  
  -- Calcular progresso
  IF total_lessons > 0 THEN
    new_progress := (completed_lessons::DECIMAL / total_lessons) * 100;
  ELSE
    new_progress := 0;
  END IF;
  
  -- Atualizar enrollment
  UPDATE enrollments
  SET 
    progress_percentage = new_progress,
    started_at = COALESCE(started_at, NOW()),
    completed_at = CASE 
      WHEN new_progress >= 100 THEN COALESCE(completed_at, NOW())
      ELSE NULL
    END
  WHERE id = NEW.enrollment_id;
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_calculate_progress
AFTER INSERT OR UPDATE ON lesson_progress
FOR EACH ROW
WHEN (NEW.status = 'completed')
EXECUTE FUNCTION calculate_enrollment_progress();
```

### Auto-update XP and level

```sql
CREATE OR REPLACE FUNCTION award_xp(
  p_user_id UUID,
  p_xp_amount INTEGER
) RETURNS void AS $$
DECLARE
  v_current_xp INTEGER;
  v_new_total_xp INTEGER;
  v_new_level INTEGER;
BEGIN
  -- Buscar XP atual
  SELECT total_xp INTO v_current_xp
  FROM user_xp
  WHERE user_id = p_user_id;
  
  -- Calcular novo total
  v_new_total_xp := v_current_xp + p_xp_amount;
  
  -- Calcular novo nível (curva logarítmica)
  v_new_level := FLOOR(1 + (LN(v_new_total_xp + 1) / LN(2)));
  
  -- Atualizar
  UPDATE user_xp
  SET 
    total_xp = v_new_total_xp,
    level = v_new_level,
    current_level_xp = v_new_total_xp - POWER(2, v_new_level - 1),
    next_level_xp = POWER(2, v_new_level) - POWER(2, v_new_level - 1),
    updated_at = NOW()
  WHERE user_id = p_user_id;
END;
$$ LANGUAGE plpgsql;

-- Trigger para dar XP ao completar lição
CREATE OR REPLACE FUNCTION give_lesson_xp()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.status = 'completed' AND (OLD.status IS NULL OR OLD.status != 'completed') THEN
    PERFORM award_xp(
      (SELECT user_id FROM enrollments WHERE id = NEW.enrollment_id),
      50 -- XP por lição
    );
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_lesson_xp
AFTER UPDATE ON lesson_progress
FOR EACH ROW
EXECUTE FUNCTION give_lesson_xp();
```

### Update streak on activity ⭐ NOVO

```sql
CREATE OR REPLACE FUNCTION update_user_streak()
RETURNS TRIGGER AS $$
DECLARE
  v_user_id UUID;
  v_last_activity DATE;
  v_current_streak INTEGER;
  v_longest_streak INTEGER;
BEGIN
  -- Pegar user_id do enrollment
  SELECT user_id INTO v_user_id
  FROM enrollments
  WHERE id = NEW.enrollment_id;
  
  -- Buscar streak atual
  SELECT last_activity_date, current_streak, longest_streak
  INTO v_last_activity, v_current_streak, v_longest_streak
  FROM user_streaks
  WHERE user_id = v_user_id;
  
  -- Lógica de streak
  IF v_last_activity IS NULL THEN
    -- Primeira atividade
    INSERT INTO user_streaks (user_id, current_streak, longest_streak, last_activity_date)
    VALUES (v_user_id, 1, 1, CURRENT_DATE);
  ELSIF v_last_activity = CURRENT_DATE THEN
    -- Já estudou hoje, não faz nada
    RETURN NEW;
  ELSIF v_last_activity = CURRENT_DATE - 1 THEN
    -- Estudou ontem, incrementa streak
    UPDATE user_streaks
    SET 
      current_streak = current_streak + 1,
      longest_streak = GREATEST(longest_streak, current_streak + 1),
      last_activity_date = CURRENT_DATE,
      updated_at = NOW()
    WHERE user_id = v_user_id;
    
    -- Dar XP bônus se completou 7 dias
    IF (v_current_streak + 1) % 7 = 0 THEN
      PERFORM award_xp(v_user_id, 300);
    END IF;
  ELSE
    -- Quebrou a sequência
    UPDATE user_streaks
    SET 
      current_streak = 1,
      last_activity_date = CURRENT_DATE,
      updated_at = NOW()
    WHERE user_id = v_user_id;
  END IF;
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_update_streak
AFTER INSERT OR UPDATE ON lesson_progress
FOR EACH ROW
WHEN (NEW.status = 'completed')
EXECUTE FUNCTION update_user_streak();
```

```typescript
        resource_id UUID,

    `);
```

## 📊 Queries Importantes

### Top 10 cursos mais populares

```sql
SELECT 
  c.id,
  c.title,
  COUNT(e.id) as total_enrollments,
  AVG(e.progress_percentage) as avg_progress,
  COUNT(CASE WHEN e.completed_at IS NOT NULL THEN 1 END) as completions,
  (COUNT(CASE WHEN e.completed_at IS NOT NULL THEN 1 END)::FLOAT / 
   NULLIF(COUNT(e.id), 0) * 100) as completion_rate
FROM courses c
LEFT JOIN enrollments e ON c.id = e.course_id
WHERE c.status = 'published'
GROUP BY c.id, c.title
ORDER BY total_enrollments DESC
LIMIT 10;
```

### Progresso de um usuário

```sql
SELECT 
  c.title as course,
  e.progress_percentage,
  e.enrolled_at,
  e.completed_at,
  COUNT(lp.id) as lessons_completed,
  MAX(lp.completed_at) as last_activity
FROM enrollments e
JOIN courses c ON e.course_id = c.id
LEFT JOIN lesson_progress lp ON e.id = lp.enrollment_id AND lp.status = 'completed'
WHERE e.user_id = $1
GROUP BY c.id, c.title, e.progress_percentage, e.enrolled_at, e.completed_at
ORDER BY e.enrolled_at DESC;
```

### Leaderboard global

```sql
SELECT 
  u.id,
  u.name,
  u.avatar_url,
  ux.total_xp,
  ux.level,
  COUNT(DISTINCT ua.id) as badges_count,
  ROW_NUMBER() OVER (ORDER BY ux.total_xp DESC) as rank
FROM users u
JOIN user_xp ux ON u.id = ux.user_id
LEFT JOIN user_achievements ua ON u.id = ua.user_id
WHERE u.is_active = true AND u.deleted_at IS NULL
GROUP BY u.id, u.name, u.avatar_url, ux.total_xp, ux.level
ORDER BY ux.total_xp DESC
LIMIT 100;
```

### Compliance: Audit trail de um usuário ⭐ NOVO

```sql
SELECT 
  al.timestamp,
  al.event_type,
  u2.name as performed_by,
```

### Sync status overview ⭐ NOVO

```sql
SELECT 
  entity_type,
  COUNT(*) as total_entities,
  COUNT(CASE WHEN last_synced_at > NOW() - INTERVAL '1 hour' THEN 1 END) as synced_last_hour,
  COUNT(CASE WHEN last_synced_at < NOW() - INTERVAL '24 hours' THEN 1 END) as stale_entities,
  AVG(EXTRACT(EPOCH FROM (NOW() - last_synced_at))) as avg_sync_lag_seconds
FROM sync_status
GROUP BY entity_type;
```

## 🔒 Security Considerations

### Particionamento

```sql
-- Audit logs por mês (já implementado acima)
-- Criar partições futuras automaticamente via cron:

CREATE OR REPLACE FUNCTION create_audit_log_partition()
RETURNS void AS $$
DECLARE
  partition_date DATE;
  partition_name TEXT;
BEGIN
  partition_date := DATE_TRUNC('month', NOW() + INTERVAL '1 month');
  partition_name := 'audit_logs_y' || TO_CHAR(partition_date, 'YYYY') || 
                   'm' || TO_CHAR(partition_date, 'MM');
  
  EXECUTE format(
    'CREATE TABLE IF NOT EXISTS %I PARTITION OF audit_logs 
     FOR VALUES FROM (%L) TO (%L)',
    partition_name,
    partition_date,
    partition_date + INTERVAL '1 month'
  );
END;
$$ LANGUAGE plpgsql;

-- Executar mensalmente via cron
```
