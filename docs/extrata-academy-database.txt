# 🗄️ EXTRATA ACADEMY - Database Design

## 📊 Overview

**Database**: PostgreSQL 17.x  
**ORM**: TypeORM ou Prisma  
**Naming Convention**: snake_case  
**Primary Keys**: UUID v4  
**Timestamps**: UTC timezone

-- Atualizar updated_at automaticamente
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_users_updated_at BEFORE UPDATE ON users
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_courses_updated_at BEFORE UPDATE ON courses
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_modules_updated_at BEFORE UPDATE ON modules
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_lessons_updated_at BEFORE UPDATE ON lessons
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- Calcular progresso do curso automaticamente
CREATE OR REPLACE FUNCTION calculate_course_progress()
RETURNS TRIGGER AS $$
DECLARE
  total_lessons INTEGER;
  completed_lessons INTEGER;
  new_progress DECIMAL(5,2);
BEGIN
  -- Conta lições obrigatórias
  SELECT COUNT(*) INTO total_lessons
  FROM lessons l
  JOIN modules m ON l.module_id = m.id
  WHERE m.course_id = (
    SELECT e.course_id FROM enrollments e WHERE e.id = NEW.enrollment_id
  ) AND l.is_mandatory = true;

  -- Conta lições completadas
  SELECT COUNT(*) INTO completed_lessons
  FROM lesson_progress lp
  WHERE lp.enrollment_id = NEW.enrollment_id
    AND lp.status = 'completed';

  -- Calcula novo progresso
  IF total_lessons > 0 THEN
    new_progress := (completed_lessons::DECIMAL / total_lessons) * 100;
  ELSE
    new_progress := 0;
  END IF;

  -- Atualiza enrollment
  UPDATE enrollments
  SET 
    progress_percentage = new_progress,
    completed_at = CASE WHEN new_progress >= 100 THEN NOW() ELSE NULL END
  WHERE id = NEW.enrollment_id;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_calculate_progress
AFTER INSERT OR UPDATE ON lesson_progress
FOR EACH ROW EXECUTE FUNCTION calculate_course_progress();

-- Atualizar streak de aprendizado
CREATE OR REPLACE FUNCTION update_learning_streak()
RETURNS TRIGGER AS $$
DECLARE
  last_date DATE;
  current_date DATE := CURRENT_DATE;
BEGIN
  -- Busca última atividade
  SELECT last_activity_date INTO last_date
  FROM learning_streaks
  WHERE user_id = NEW.user_id;

  IF last_date IS NULL THEN
    -- Primeira atividade
    INSERT INTO learning_streaks (user_id, current_streak, longest_streak, last_activity_date)
    VALUES (NEW.user_id, 1, 1, current_date);
  ELSIF last_date = current_date THEN
    -- Já teve atividade hoje, nada a fazer
    RETURN NEW;
  ELSIF last_date = current_date - INTERVAL '1 day' THEN
    -- Dia consecutivo, incrementa streak
    UPDATE learning_streaks
    SET 
      current_streak = current_streak + 1,
      longest_streak = GREATEST(longest_streak, current_streak + 1),
      last_activity_date = current_date
    WHERE user_id = NEW.user_id;
  ELSE
    -- Quebrou a sequência, reseta
    UPDATE learning_streaks
    SET 
      current_streak = 1,
      last_activity_date = current_date
    WHERE user_id = NEW.user_id;
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_update_streak
AFTER INSERT ON lesson_progress
FOR EACH ROW
WHEN (NEW.status = 'completed')
EXECUTE FUNCTION update_learning_streak();

## 📊 Queries Importantes

### Top 10 cursos mais populares

```sql
SELECT 
  c.id,
  c.title,
  COUNT(e.id) as total_enrollments,
  AVG(e.progress_percentage) as avg_progress,
  COUNT(CASE WHEN e.completed_at IS NOT NULL THEN 1 END) as completions
FROM courses c
LEFT JOIN enrollments e ON c.id = e.course_id
WHERE c.status = 'published'
GROUP BY c.id, c.title
ORDER BY total_enrollments DESC
LIMIT 10;
```

### Progresso de um usuário

```sql
SELECT 
  c.title as course,
  e.progress_percentage,
  e.enrolled_at,
  e.completed_at,
  COUNT(lp.id) as lessons_completed,
  MAX(lp.completed_at) as last_activity
FROM enrollments e
JOIN courses c ON e.course_id = c.id
LEFT JOIN lesson_progress lp ON e.id = lp.enrollment_id AND lp.status = 'completed'
WHERE e.user_id = $1
GROUP BY c.id, c.title, e.progress_percentage, e.enrolled_at, e.completed_at
ORDER BY e.enrolled_at DESC;
```

### Leaderboard global

```sql
SELECT 
  u.id,
  u.name,
  u.avatar_url,
  ux.total_xp,
  ux.level,
  COUNT(DISTINCT ua.id) as badges_count,
  ROW_NUMBER() OVER (ORDER BY ux.total_xp DESC) as rank
FROM users u
JOIN user_xp ux ON u.id = ux.user_id
LEFT JOIN user_achievements ua ON u.id = ua.user_id
WHERE u.is_active = true
GROUP BY u.id, u.name, u.avatar_url, ux.total_xp, ux.level
ORDER BY ux.total_xp DESC
LIMIT 100;
```

## 🔒 Security Considerations

1. **Row Level Security (RLS)**: Considerar para multi-tenancy
2. **Prepared Statements**: Sempre usar (TypeORM já faz)
3. **Encryption**: Dados sensíveis em rest (pgcrypto)
4. **Audit Log**: Tabela separada para auditoria
5. **Backups**: Daily automático + point-in-time recovery
