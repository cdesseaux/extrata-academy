# 🔒 EXTRATA ACADEMY - Compliance e LGPD

## 📋 Overview

Este documento estabelece as políticas e procedimentos para garantir **compliance total com a LGPD** (Lei Geral de Proteção de Dados - Lei 13.709/2018) no Extrata Academy.

**Objetivo**: Proteger os dados pessoais dos usuários e garantir conformidade legal.

---

## 🎯 Princípios da LGPD

O Extrata Academy segue os 10 princípios estabelecidos pela LGPD:

1. ✅ **Finalidade** - Coleta para propósitos legítimos e específicos
2. ✅ **Adequação** - Compatibilidade com finalidades informadas
3. ✅ **Necessidade** - Limitação ao mínimo necessário
4. ✅ **Livre acesso** - Garantia de consulta facilitada
5. ✅ **Qualidade dos dados** - Exatidão e atualização
6. ✅ **Transparência** - Informações claras e acessíveis
7. ✅ **Segurança** - Medidas técnicas e administrativas
8. ✅ **Prevenção** - Medidas para evitar danos
9. ✅ **Não discriminação** - Impossibilidade de uso discriminatório
10. ✅ **Responsabilização** - Demonstração de conformidade

```typescript
export interface ConsentRecord {
  user_id: string;
  consent_type: 'terms' | 'privacy' | 'notifications' | 'marketing';
  granted: boolean;
  version: string; // Versão do documento aceito
  timestamp: Date;
  ip_address: string;
  user_agent: string;
}

@Injectable()
export class ConsentService {
  async recordConsent(data: ConsentRecord) {
    // Salvar no banco para auditoria
    await this.db.consents.create(data);
    
    // Também logar para imutabilidade
    this.logger.log('consent_granted', data);
  }

  async checkConsent(userId: string, type: string): Promise<boolean> {
    const latest = await this.db.consents.findOne({
      where: { user_id: userId, consent_type: type },
      order: { timestamp: 'DESC' },
    });

    return latest?.granted ?? false;
  }

  async revokeConsent(userId: string, type: string) {
    await this.recordConsent({
      user_id: userId,
      consent_type: type as any,
      granted: false,
      version: CURRENT_TERMS_VERSION,
      timestamp: new Date(),
      ip_address: '...', // From request
      user_agent: '...', // From request
    });
  }
}
```

```typescript
// src/compliance/rights.service.ts
@Injectable()
export class DataSubjectRightsService {
  // Direito de acesso (Art. 18, II)
  async exportUserData(userId: string): Promise<string> {
    const user = await this.usersRepo.findOne(userId);
    const enrollments = await this.enrollmentsRepo.find({ user_id: userId });
    const progress = await this.progressRepo.find({ user_id: userId });
    const certificates = await this.certificatesRepo.find({ user_id: userId });
    const consents = await this.consentsRepo.find({ user_id: userId });

    const exportData = {
      user,
      enrollments,
      progress,
      certificates,
      consents,
      exported_at: new Date().toISOString(),
    };

    // Salvar em S3 e enviar link por email
    const fileUrl = await this.s3.upload(
      `exports/${userId}-${Date.now()}.json`,
      JSON.stringify(exportData, null, 2),
    );

    await this.emailService.send({
      to: user.email,
      subject: 'Seus dados do Extrata Academy',
      body: `Seus dados estão disponíveis em: ${fileUrl}`,
    });

    return fileUrl;
  }

  // Direito de portabilidade (Art. 18, V)
  async exportPortableData(userId: string): Promise<object> {
    // Formato legível por máquina (JSON)
    return {
      user: await this.getUserPortableData(userId),
      courses: await this.getCoursesPortableData(userId),
      achievements: await this.getAchievementsPortableData(userId),
    };
  }

  // Direito de eliminação (Art. 18, VI)
  async deleteUserData(userId: string) {
    const user = await this.usersRepo.findOne(userId);

    // 1. Anonimizar dados pessoais
    await this.usersRepo.update(userId, {
      name: `Usuário Excluído ${userId.slice(0, 8)}`,
      email: `deleted-${userId}@anonymized.local`,
      cpf: null,
      phone: null,
      avatar_url: null,
      is_active: false,
      deleted_at: new Date(),
    });

    // 2. Manter dados necessários por lei (certificados com CPF)
    // Esses não são excluídos

    // 3. Excluir progresso e dados de uso
    await this.progressRepo.delete({ user_id: userId });
    
    // 4. Logar exclusão para auditoria
    this.logger.log('user_deleted', {
      user_id: userId,
      email: user.email,
      deleted_at: new Date(),
      deleted_by: userId,
    });

    // 5. Notificar DPO
    await this.notifyDPO({
      event: 'USER_DELETION',
      user_id: userId,
      timestamp: new Date(),
    });
  }

  // Direito de correção (Art. 18, III)
  async correctUserData(userId: string, corrections: Partial<User>) {
    await this.usersRepo.update(userId, corrections);
    
    // Logar correção
    this.logger.log('data_corrected', {
      user_id: userId,
      fields: Object.keys(corrections),
      timestamp: new Date(),
    });
  }
}
```

```typescript
// src/compliance/audit-logger.ts
export interface AuditLogEntry {
  timestamp: Date;
  event_type: string;
  user_id: string;
  resource_type: string;
  resource_id: string;
  action: 'CREATE' | 'READ' | 'UPDATE' | 'DELETE';
  ip_address: string;
  user_agent: string;
  metadata?: Record<string, any>;
}

@Injectable()
export class AuditLogger {
  async log(entry: Omit<AuditLogEntry, 'timestamp'>) {
    // Salvar em tabela específica
    await this.db.audit_logs.create({
      ...entry,
      timestamp: new Date(),
    });

    // Também enviar para sistema de logs centralizado
    this.logger.info('audit', entry);

    // Se for ação sensível, alertar
    if (this.isSensitiveAction(entry)) {
      await this.alertSecurityTeam(entry);
    }
  }

  private isSensitiveAction(entry: AuditLogEntry): boolean {
    return [
      'user_deleted',
      'role_changed',
      'admin_access',
      'data_exported',
    ].includes(entry.event_type);
  }
}

// Uso em controllers
@Controller('users')
export class UsersController {
  @Delete(':id')
  @UseGuards(AdminGuard)
  async deleteUser(
    @Param('id') userId: string,
    @CurrentUser() admin: User,
    @Ip() ip: string,
  ) {
    await this.usersService.delete(userId);
    
    // Logar para auditoria
    await this.auditLogger.log({
      event_type: 'user_deleted',
      user_id: admin.id,
      resource_type: 'user',
      resource_id: userId,
      action: 'DELETE',
      ip_address: ip,
      user_agent: req.headers['user-agent'],
    });
  }
}
```

```sql
-- Query para relatório de acessos aos dados de um usuário
SELECT 
  al.timestamp,
  al.event_type,
  u.name as accessed_by,
  al.action,
  al.ip_address
FROM audit_logs al
JOIN users u ON al.user_id = u.id
WHERE al.resource_type = 'user'
  AND al.resource_id = $target_user_id
  AND al.timestamp >= NOW() - INTERVAL '90 days'
ORDER BY al.timestamp DESC;
```

## 🚨 Incident Response Plan

### Data Breach Protocol

```typescript
// src/compliance/incident-response.ts
export class IncidentResponseService {
  async reportDataBreach(incident: DataBreachIncident) {
    // 1. Contenção imediata
    await this.containBreach(incident);
    
    // 2. Avaliar severidade
    const severity = await this.assessSeverity(incident);
    
    // 3. Notificar DPO
    await this.notifyDPO(incident, severity);
    
    // 4. Se alta severidade, notificar ANPD (72h)
    if (severity === 'HIGH' || severity === 'CRITICAL') {
      await this.notifyANPD(incident);
    }
    
    // 5. Notificar usuários afetados
    await this.notifyAffectedUsers(incident);
    
    // 6. Documentar incidente
    await this.documentIncident(incident);
    
    // 7. Implementar correções
    await this.implementCorrections(incident);
  }

  private async notifyANPD(incident: DataBreachIncident) {
    // Prazo: 72 horas após conhecimento do incidente
    const report = {
      incident_date: incident.detected_at,
      nature_of_breach: incident.type,
      affected_data_categories: incident.affected_data,
      number_of_affected_users: incident.affected_users_count,
      likely_consequences: incident.consequences,
      measures_taken: incident.remediation,
      dpo_contact: DPO_CONTACT,
    };

    // Enviar para ANPD via formulário oficial
    await this.submitToANPD(report);
  }

  private async notifyAffectedUsers(incident: DataBreachIncident) {
    const users = await this.getAffectedUsers(incident);
    
    for (const user of users) {
      await this.emailService.send({
        to: user.email,
        subject: 'IMPORTANTE: Incidente de Segurança - Extrata Academy',
        body: this.generateBreachNotificationEmail(user, incident),
      });
    }
  }
}
```
