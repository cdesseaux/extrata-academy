# 📊 EXTRATA ACADEMY - Observability & Monitoring

## 📋 Overview

Este documento define a estratégia completa de **observabilidade** para o Extrata Academy, cobrindo:

- 📈 **Métricas** - O que está acontecendo (Prometheus/Grafana)
- 📝 **Logs** - Por que está acontecendo (Winston/Loki)
- 🔍 **Tracing** - Como está acontecendo (Jaeger/Zipkin)
- 🚨 **Alertas** - Quando algo está errado (Alertmanager)

**Objetivo**: Detectar, diagnosticar e resolver problemas proativamente.

### Instrumentação no Código

```typescript
// src/common/interceptors/metrics.interceptor.ts
@Injectable()
export class MetricsInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest();
    const { method, route } = request;
    
    const end = httpRequestDuration.startTimer();
    
    return next.handle().pipe(
      tap(() => {
        const response = context.switchToHttp().getResponse();
        const { statusCode } = response;
        
        // Record duration
        end({ method, route: route.path, status_code: statusCode });
        
        // Count request
        httpRequestTotal.inc({ method, route: route.path, status_code: statusCode });
      }),
      catchError((error) => {
        end({ method, route: route.path, status_code: error.status || 500 });
        httpRequestTotal.inc({ method, route: route.path, status_code: error.status || 500 });
        throw error;
      }),
    );
  }
}

// Aplicar globalmente
app.useGlobalInterceptors(new MetricsInterceptor());
```

## 📝 Logs (Winston + Loki)

### Estrutura de Logs

```typescript
// src/common/logger/logger.service.ts
import * as winston from 'winston';
import LokiTransport from 'winston-loki';

export const logger = winston.createLogger({
  level: process.env.LOG_LEVEL || 'info',
  format: winston.format.combine(
    winston.format.timestamp(),
    winston.format.errors({ stack: true }),
    winston.format.json(),
  ),
  defaultMeta: {
    service: 'academy-backend',
    environment: process.env.NODE_ENV,
    version: process.env.APP_VERSION,
  },
  transports: [
    // Console (desenvolvimento)
    new winston.transports.Console({
      format: winston.format.combine(
        winston.format.colorize(),
        winston.format.simple(),
      ),
    }),
    
    // Arquivo (produção)
    new winston.transports.File({
      filename: 'logs/error.log',
      level: 'error',
      maxsize: 10485760, // 10MB
      maxFiles: 5,
    }),
    
    new winston.transports.File({
      filename: 'logs/combined.log',
      maxsize: 10485760,
      maxFiles: 10,
    }),
    
    // Loki (produção)
    new LokiTransport({
      host: process.env.LOKI_URL,
      labels: { app: 'academy-backend' },
      json: true,
      format: winston.format.json(),
      replaceTimestamp: true,
      onConnectionError: (err) => console.error(err),
    }),
  ],
});

// Contextualized logging
export class LoggerService {
  private context: string;

  constructor(context: string) {
    this.context = context;
  }

  log(message: string, meta?: any) {
    logger.info(message, { context: this.context, ...meta });
  }

  error(message: string, trace?: string, meta?: any) {
    logger.error(message, {
      context: this.context,
      stack: trace,
      ...meta,
    });
  }

  warn(message: string, meta?: any) {
    logger.warn(message, { context: this.context, ...meta });
  }

  debug(message: string, meta?: any) {
    logger.debug(message, { context: this.context, ...meta });
  }

  // Structured logging for specific events
  logUserAction(userId: string, action: string, details: any) {
    logger.info('user_action', {
      context: this.context,
      user_id: userId,
      action,
      details,
      timestamp: new Date().toISOString(),
    });
  }

  logAPICall(endpoint: string, duration: number, statusCode: number) {
    logger.info('api_call', {
      context: this.context,
      endpoint,
      duration_ms: duration,
      status_code: statusCode,
      timestamp: new Date().toISOString(),
    });
  }
}
```

### Uso nos Services

```typescript
// src/courses/courses.service.ts
@Injectable()
export class CoursesService {
  private readonly logger = new LoggerService(CoursesService.name);

  async createCourse(data: CreateCourseDto, userId: string) {
    this.logger.log('Creating new course', { title: data.title, created_by: userId });

    try {
      const course = await this.coursesRepo.create(data);
      
      this.logger.log('Course created successfully', {
        course_id: course.id,
        title: course.title,
      });

      // Log para auditoria
      this.logger.logUserAction(userId, 'course_created', {
        course_id: course.id,
        title: course.title,
      });

      return course;
    } catch (error) {
      this.logger.error('Failed to create course', error.stack, {
        data,
        user_id: userId,
      });
      throw error;
    }
  }
}

### Configuração do Tracer

```typescript
// src/common/tracing/tracer.ts
import { initTracer } from 'jaeger-client';
import * as opentracing from 'opentracing';

export function initializeTracing() {
  const config = {
    serviceName: 'academy-backend',
    sampler: {
      type: 'probabilistic',
      param: process.env.NODE_ENV === 'production' ? 0.1 : 1.0, // 10% em prod
    },
    reporter: {
      logSpans: true,
      agentHost: process.env.JAEGER_AGENT_HOST || 'localhost',
      agentPort: 6831,
    },
  };

  const options = {
    tags: {
      'academy.version': process.env.APP_VERSION,
      'academy.environment': process.env.NODE_ENV,
    },
    logger: {
      info: (msg) => logger.info(msg),
      error: (msg) => logger.error(msg),
    },
  };

  const tracer = initTracer(config, options);
  opentracing.initGlobalTracer(tracer);
  
  return tracer;
}

### Instrumentação de Requests

```typescript
// src/common/interceptors/tracing.interceptor.ts
@Injectable()
export class TracingInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest();
    const tracer = opentracing.globalTracer();
    
    // Extract parent span from headers (if exists)
    const parentSpanContext = tracer.extract(
      opentracing.FORMAT_HTTP_HEADERS,
      request.headers,
    );
    
    // Create span
    const span = tracer.startSpan(request.route.path, {
      childOf: parentSpanContext,
      tags: {
        [opentracing.Tags.SPAN_KIND]: opentracing.Tags.SPAN_KIND_RPC_SERVER,
        [opentracing.Tags.HTTP_METHOD]: request.method,
        [opentracing.Tags.HTTP_URL]: request.url,
      },
    });
    
    // Attach span to request
    request.span = span;
    
    return next.handle().pipe(
      tap(() => {
        const response = context.switchToHttp().getResponse();
        span.setTag(opentracing.Tags.HTTP_STATUS_CODE, response.statusCode);
        span.finish();
      }),
      catchError((error) => {
        span.setTag(opentracing.Tags.ERROR, true);
        span.log({
          event: 'error',
          message: error.message,
          stack: error.stack,
        });
        span.finish();
        throw error;
      }),
    );
  }
}
```

### Tracing Database Queries

```typescript
// src/common/database/traced-repository.ts
export class TracedRepository<T> extends Repository<T> {
  async find(options?: FindManyOptions<T>): Promise<T[]> {
    const span = opentracing.globalTracer().startSpan('database.find', {
      tags: {
        'db.type': 'postgresql',
        'db.table': this.metadata.tableName,
      },
    });

    try {
      const result = await super.find(options);
      span.setTag('db.rows_returned', result.length);
      return result;
    } catch (error) {
      span.setTag(opentracing.Tags.ERROR, true);
      span.log({ event: 'error', message: error.message });
      throw error;
    } finally {
      span.finish();
    }
  }
}
```

### Tracing External API Calls

```typescript
// src/integration/extrata-client.ts
export class ExtrataApiClient {
  async getUser(keycloakId: string) {
    const span = opentracing.globalTracer().startSpan('extrata_api.get_user', {
      tags: {
        'http.method': 'GET',
        'http.url': `${EXTRATA_API_URL}/users/${keycloakId}`,
      },
    });

    try {
      const response = await fetch(`${EXTRATA_API_URL}/users/${keycloakId}`, {
        headers: {
          'Authorization': `Bearer ${this.accessToken}`,
          // Inject trace context
          ...opentracing.globalTracer().inject(span, opentracing.FORMAT_HTTP_HEADERS, {}),
        },
      });

      span.setTag('http.status_code', response.status);
      
      const data = await response.json();
      return data;
    } catch (error) {
      span.setTag(opentracing.Tags.ERROR, true);
      span.log({ event: 'error', message: error.message });
      throw error;
    } finally {
      span.finish();
    }
  }
}

## 🚨 Alertas (Prometheus Alertmanager)

### Regras de Alerta

```yaml
# prometheus/alerts.yml
groups:
  - name: academy_critical
    interval: 30s
    rules:
      # High Error Rate
      - alert: HighErrorRate
        expr: rate(http_requests_total{status_code=~"5.."}[5m]) > 0.01
        for: 2m
        labels:
          severity: critical
          team: backend
        annotations:
          summary: "High error rate detected"
          description: "Error rate is {{ $value | humanizePercentage }} (threshold: 1%)"
          
      # API Latency
      - alert: HighAPILatency
        expr: histogram_quantile(0.95, rate(http_request_duration_seconds_bucket[5m])) > 0.5
        for: 5m
        labels:
          severity: warning
          team: backend
        annotations:
          summary: "API latency is high"
          description: "P95 latency is {{ $value }}s (threshold: 500ms)"
          
      # Database Issues
      - alert: HighDatabaseLatency
        expr: histogram_quantile(0.95, rate(database_query_duration_seconds_bucket[5m])) > 0.1
        for: 3m
        labels:
          severity: warning
          team: backend
        annotations:
          summary: "Database queries are slow"
          description: "P95 query time is {{ $value }}s (threshold: 100ms)"
          
      - alert: DatabaseConnectionPoolExhausted
        expr: database_connections_active / database_connections_max > 0.9
        for: 2m
        labels:
          severity: critical
          team: backend
        annotations:
          summary: "Database connection pool nearly exhausted"
          description: "{{ $value | humanizePercentage }} of connections in use"
          
      # Service Down
      - alert: ServiceDown
        expr: up{job="academy-backend"} == 0
        for: 1m
        labels:
          severity: critical
          team: sre
        annotations:
          summary: "Academy backend is down"
          description: "Service has been down for more than 1 minute"
          
      # Video Streaming Issues
      - alert: HighVideoBuffering
        expr: rate(video_buffering_events_total[5m]) > 0.1

        labels:
          severity: warning
          team: backend

## 📚 Runbooks

### Runbook: High Error Rate

```markdown
# High Error Rate (>1%)

## Severity: CRITICAL

## Description
The API error rate has exceeded 1% for more than 2 minutes.

## Impact
Users are experiencing errors when using the platform.

## Diagnosis

1. Check Grafana dashboard: https://grafana.academy/d/errors
2. Look at error logs:
   ```bash
   kubectl logs -f deployment/academy-backend --tail=100 | grep "ERROR"
   ```
3. Check Jaeger for failed traces:
   ```
   https://jaeger.academy/search?service=academy-backend&tags=error:true
   ```

## Possible Causes

- Database connection issues
- External API (Extrata) down
- Keycloak authentication issues
- Code bug introduced in recent deploy

## Resolution Steps

### If Database Issue:
```bash
# Check database health
kubectl exec -it postgres-0 -- psql -U academy -c "SELECT 1;"

# Check connection pool
SELECT count(*) FROM pg_stat_activity WHERE datname = 'academy';

# If needed, restart pods to reset connections
kubectl rollout restart deployment/academy-backend
```

### If Extrata API Issue:
```bash
# Check Extrata API status
curl https://api.extrata.gov.br/health

# Enable fallback mode (use cached data)

### Runbook: Database Connection Pool Exhausted

```markdown
# Database Connection Pool Exhausted

## Severity: CRITICAL

## Description
90%+ of database connections are in use.

## Diagnosis

```sql
-- Check active connections
SELECT count(*), state 
FROM pg_stat_activity 
WHERE datname = 'academy' 
GROUP BY state;

-- Check long-running queries
SELECT pid, now() - pg_stat_activity.query_start AS duration, query 
FROM pg_stat_activity 
WHERE state = 'active' 
  AND now() - pg_stat_activity.query_start > interval '1 minute';
```

## Resolution

```bash
# Increase connection pool size (temporary)
kubectl set env deployment/academy-backend DB_POOL_SIZE=30

# Kill long-running queries (if safe)
SELECT pg_terminate_backend(pid) FROM pg_stat_activity 
WHERE state = 'active' 
  AND now() - pg_stat_activity.query_start > interval '5 minutes';

# Restart pods to reset connections
kubectl rollout restart deployment/academy-backend
```

## Prevention

- Review slow queries and add indexes
- Implement query timeout
- Use read replicas for heavy queries
```
