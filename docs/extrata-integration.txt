# 🔗 EXTRATA ACADEMY - Integração com Sistema Extrata

## 📋 Overview

Este documento especifica como o **Extrata Academy** se integra ao **sistema Extrata** (plataforma principal de gestão governamental), garantindo:

- ✅ **Single Sign-On (SSO)** - Um login para ambos os sistemas
- ✅ **Sincronização de dados** - Usuários, municípios, roles
- ✅ **Context sharing** - Navegação fluida entre sistemas
- ✅ **Webhooks** - Eventos em tempo real
- ✅ **Deep linking** - Links diretos entre plataformas

### Implementação do SSO

#### Backend (NestJS)

```typescript
// src/auth/auth.controller.ts
@Controller('auth')
export class AuthController {
  constructor(private readonly authService: AuthService) {}

  @Get('login')
  login(@Res() res: Response) {
    const keycloakAuthUrl = this.authService.getKeycloakAuthUrl();
    return res.redirect(keycloakAuthUrl);
  }

  @Get('callback')
  async callback(
    @Query('code') code: string,
    @Res() res: Response,
  ) {
    try {
      // 1. Trocar code por tokens
      const tokens = await this.authService.exchangeCodeForTokens(code);
      
      // 2. Decodificar token e extrair user info
      const userInfo = await this.authService.getUserInfo(tokens.access_token);
      
      // 3. Buscar dados adicionais do Extrata Main
      const extraData = await this.authService.fetchExtrataUserData(userInfo.sub);
      
      // 4. Sincronizar/criar usuário local
      const user = await this.authService.syncUser({
        keycloak_id: userInfo.sub,
        email: userInfo.email,
        name: userInfo.name,
        role: this.authService.mapRole(userInfo.realm_access?.roles),
        municipality_id: extraData.municipality_id,
        metadata: extraData.metadata,
      });
      
      // 5. Criar sessão Academy
      const sessionToken = await this.authService.createSession(user);
      
      // 6. Redirecionar para dashboard
      res.cookie('academy_session', sessionToken, {
        httpOnly: true,
        secure: process.env.NODE_ENV === 'production',
        sameSite: 'lax',
        maxAge: 7 * 24 * 60 * 60 * 1000, // 7 dias
      });
      
      return res.redirect('/dashboard');
    } catch (error) {
      return res.redirect('/error?message=auth_failed');
    }
  }

  @Post('refresh')
  async refresh(@Body('refresh_token') refreshToken: string) {
    const newTokens = await this.authService.refreshTokens(refreshToken);
    return newTokens;
  }

  @Post('logout')
  @UseGuards(JwtAuthGuard)
  async logout(
    @CurrentUser() user: User,
    @Res() res: Response,
  ) {
    await this.authService.revokeSession(user.id);
    
    // Logout do Keycloak também
    const logoutUrl = await this.authService.getKeycloakLogoutUrl();
    
    res.clearCookie('academy_session');
    return res.redirect(logoutUrl);
  }
}
```

```typescript
// src/auth/role-mapper.ts
export class RoleMapper {
  private static readonly ROLE_MAPPING = {
    // Roles do Keycloak → Roles do Academy
    'extrata_admin': 'ADMIN',
    'extrata_gestor': 'INSTRUCTOR',
    'extrata_usuario': 'STUDENT',
    'extrata_auditor': 'AUDITOR',
  };

  static mapKeycloakRolesToAcademy(keycloakRoles: string[]): string {
    // Prioridade: ADMIN > INSTRUCTOR > AUDITOR > STUDENT
    if (keycloakRoles.includes('extrata_admin')) return 'ADMIN';
    if (keycloakRoles.includes('extrata_gestor')) return 'INSTRUCTOR';
    if (keycloakRoles.includes('extrata_auditor')) return 'AUDITOR';
    return 'STUDENT';
  }
}
```

### Service Account Authentication

```typescript
// src/integration/extrata-client.ts
export class ExtrataApiClient {
  private accessToken: string;
  private tokenExpiresAt: Date;

  async authenticate() {
    const response = await fetch(`${EXTRATA_API_URL}/oauth/token`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: new URLSearchParams({
        grant_type: 'client_credentials',
        client_id: process.env.EXTRATA_CLIENT_ID,
        client_secret: process.env.EXTRATA_CLIENT_SECRET,
        scope: 'read:users read:municipalities',
      }),
    });

    const data = await response.json();
    this.accessToken = data.access_token;
    this.tokenExpiresAt = new Date(Date.now() + data.expires_in * 1000);
  }

  async getUser(keycloakId: string) {
    await this.ensureAuthenticated();
    
    const response = await fetch(`${EXTRATA_API_URL}/api/v1/users/${keycloakId}`, {
      headers: {
        'Authorization': `Bearer ${this.accessToken}`,
        'X-API-Key': process.env.EXTRATA_API_KEY,
      },
    });

    if (!response.ok) {
      throw new Error(`Failed to fetch user: ${response.status}`);
    }

    return response.json();
  }

  private async ensureAuthenticated() {
    if (!this.accessToken || this.tokenExpiresAt < new Date()) {
      await this.authenticate();
    }
  }
}
```

## 🔔 Webhooks

### Eventos Enviados pelo Extrata

```typescript
// src/integration/webhook-handler.ts
export enum WebhookEventType {
  USER_CREATED = 'user.created',
  USER_UPDATED = 'user.updated',
  USER_DELETED = 'user.deleted',
  MUNICIPALITY_CREATED = 'municipality.created',
  MUNICIPALITY_UPDATED = 'municipality.updated',
}

@Controller('webhooks')
export class WebhookController {
  @Post('extrata')
  async handleExtrataWebhook(
    @Body() payload: WebhookPayload,
    @Headers('x-extrata-signature') signature: string,
  ) {
    // 1. Validar assinatura
    if (!this.validateSignature(payload, signature)) {
      throw new UnauthorizedException('Invalid webhook signature');
    }

    // 2. Processar evento
    try {
      await this.processWebhookEvent(payload);
      return { success: true };
    } catch (error) {
      // Extrata vai retentar se retornarmos erro
      throw new InternalServerErrorException('Failed to process webhook');
    }
  }

  private validateSignature(payload: any, signature: string): boolean {
    const hmac = crypto.createHmac('sha256', process.env.WEBHOOK_SECRET);
    hmac.update(JSON.stringify(payload));
    const expectedSignature = hmac.digest('hex');
    return crypto.timingSafeEqual(
      Buffer.from(signature),
      Buffer.from(expectedSignature),
    );
  }

  private async processWebhookEvent(payload: WebhookPayload) {
    switch (payload.event_type) {
      case WebhookEventType.USER_CREATED:
      case WebhookEventType.USER_UPDATED:
        await this.syncUserFromExtrata(payload.data.user_id);
        break;
        
      case WebhookEventType.USER_DELETED:
        await this.softDeleteUser(payload.data.user_id);
        break;
        
      case WebhookEventType.MUNICIPALITY_CREATED:
      case WebhookEventType.MUNICIPALITY_UPDATED:
        await this.syncMunicipality(payload.data.municipality_id);
        break;
    }
  }
}
```

### Retry Logic

```typescript
// src/integration/webhook-retry.ts
export class WebhookRetryService {
  async processWithRetry(payload: WebhookPayload, maxRetries = 3) {
    let attempt = 0;
    
    while (attempt < maxRetries) {
      try {
        await this.processWebhookEvent(payload);
        return;
      } catch (error) {
        attempt++;
        
        if (attempt >= maxRetries) {
          // Log erro e notifica equipe
          await this.logFailedWebhook(payload, error);
          await this.notifyTeam(payload, error);
          throw error;
        }
        
        // Exponential backoff: 1s, 2s, 4s
        const delay = Math.pow(2, attempt) * 1000;
        await this.sleep(delay);
      }
    }
  }
}
```

## 🔄 Sincronização de Dados

### Estratégias de Sync

#### 1. Real-time (via Webhooks)
- **Uso**: Dados críticos (usuários, roles)
- **Latência**: < 1 segundo
- **Implementação**: Webhooks do Extrata → Academy

#### 2. Near Real-time (via Polling)
- **Uso**: Dados menos críticos (municípios)
- **Latência**: 5-15 minutos
- **Implementação**: Cron job busca mudanças

#### 3. Batch (Scheduled)
- **Uso**: Analytics, relatórios
- **Latência**: 24 horas
- **Implementação**: Job noturno

### Implementação de Sync Service

```typescript
// src/integration/sync.service.ts
@Injectable()
export class SyncService {
  constructor(
    private extratoClient: ExtrataApiClient,
    private usersRepo: UsersRepository,
  ) {}

  // Sincronização manual de um usuário
  async syncUser(keycloakId: string) {
    const extraData = await this.extratoClient.getUser(keycloakId);
    
    await this.usersRepo.upsert({
      keycloak_id: extraData.keycloak_id,
      email: extraData.email,
      name: extraData.name,
      municipality_id: extraData.municipality.id,
      metadata: extraData.metadata,
      synced_at: new Date(),
    });
  }

  // Sincronização em batch (cron)
  @Cron('0 */15 * * * *') // A cada 15 minutos
  async syncRecentlyUpdatedUsers() {
    const since = new Date(Date.now() - 15 * 60 * 1000); // últimos 15min
    
    const updatedUsers = await this.extratoClient.getUpdatedUsers(since);
    
    for (const user of updatedUsers) {
      await this.syncUser(user.keycloak_id);
    }
  }
}
```

## 🛡️ Error Handling

### Fallback Strategies

```typescript
// src/integration/fallback.service.ts
export class IntegrationFallbackService {
  async getUserWithFallback(keycloakId: string): Promise<User> {
    try {
      // 1. Tentar buscar do Extrata Main
      return await this.extratoClient.getUser(keycloakId);
    } catch (error) {
      // 2. Se falhar, usar dados em cache
      const cachedUser = await this.cache.get(`user:${keycloakId}`);
      if (cachedUser) {
        this.logger.warn(`Using cached user data for ${keycloakId}`);
        return cachedUser;
      }
      
      // 3. Se não tiver cache, usar dados básicos do Keycloak
      const keycloakUser = await this.keycloak.getUser(keycloakId);
      return this.mapKeycloakToUser(keycloakUser);
    }
  }
}
```

### Circuit Breaker

```typescript
// src/integration/circuit-breaker.ts
export class CircuitBreaker {
  private failures = 0;
  private lastFailureTime: Date;
  private state: 'CLOSED' | 'OPEN' | 'HALF_OPEN' = 'CLOSED';

  async execute<T>(fn: () => Promise<T>): Promise<T> {
    if (this.state === 'OPEN') {
      // Se passou tempo suficiente, tenta novamente
      if (Date.now() - this.lastFailureTime.getTime() > 60000) {
        this.state = 'HALF_OPEN';
      } else {
        throw new Error('Circuit breaker is OPEN');
      }
    }

    try {
      const result = await fn();
      
      // Reset em caso de sucesso
      if (this.state === 'HALF_OPEN') {
        this.state = 'CLOSED';
        this.failures = 0;
      }
      
      return result;
    } catch (error) {
      this.failures++;
      this.lastFailureTime = new Date();
      
      // Abre circuito após 5 falhas
      if (this.failures >= 5) {
        this.state = 'OPEN';
      }
      
      throw error;
    }
  }
}
```
