## 🔧 Stack Tecnológico

### Backend
- **Framework**: NestJS 10.x (Node.js 20.x LTS)
- **Language**: TypeScript 5.x
- **ORM**: Prisma 5.x (ou TypeORM 0.3.x)
- **Database**: PostgreSQL 17.x
- **Cache**: Redis 7.x
- **Auth**: Keycloak (OAuth2/OIDC)
- **Validation**: class-validator + class-transformer
- **Documentation**: Swagger/OpenAPI 3.0

### Frontend
- **Framework**: Next.js 14.x (App Router)
- **Language**: TypeScript 5.x
- **UI Library**: React 18.x
- **Styling**: Tailwind CSS 3.x
- **Components**: shadcn/ui (Radix UI)
- **Forms**: React Hook Form + Zod
- **State**: Zustand (ou React Context)
- **HTTP Client**: TanStack Query (React Query)
- **Video Player**: Video.js ou Plyr.io

### Infrastructure
- **Containerization**: Docker + Docker Compose
- **Orchestration**: Kubernetes (futuro)
- **CI/CD**: GitHub Actions
- **Hosting**: AWS / DigitalOcean
- **CDN**: CloudFlare / CloudFront
- **Storage**: AWS S3 / Spaces
- **Monitoring**: Prometheus + Grafana
- **Logging**: Winston + ELK Stack
- **Error Tracking**: Sentry

### Exemplo: Courses Module

```typescript
// courses/courses.module.ts
@Module({
  imports: [
    DatabaseModule,
    StorageModule,
    CacheModule,
  ],
  controllers: [CoursesController],
  providers: [
    CoursesService,
    CoursesRepository,
    ModulesService,
    LessonsService,
  ],
  exports: [CoursesService],
})
export class CoursesModule {}

// courses/courses.controller.ts
@Controller('courses')
@ApiTags('Courses')
@UseGuards(JwtAuthGuard, RolesGuard)
export class CoursesController {
  constructor(private readonly coursesService: CoursesService) {}

  @Get()
  @ApiOperation({ summary: 'List all courses' })
  @ApiQuery({ name: 'category', required: false })
  @ApiResponse({ status: 200, type: [CourseDto] })
  async findAll(@Query() filters: CoursesFilterDto): Promise<CourseDto[]> {
    return this.coursesService.findAll(filters);
  }

  @Post()
  @Roles('ADMIN', 'INSTRUCTOR')
  @ApiOperation({ summary: 'Create new course' })
  @ApiBody({ type: CreateCourseDto })
  @ApiResponse({ status: 201, type: CourseDto })
  async create(@Body() dto: CreateCourseDto): Promise<CourseDto> {
    return this.coursesService.create(dto);
  }

  @Get(':id/with-progress')
  @ApiOperation({ summary: 'Get course with user progress' })
  async getCourseWithProgress(
    @Param('id') id: string,
    @CurrentUser() user: User,
  ): Promise<CourseWithProgressDto> {
    return this.coursesService.getCourseWithProgress(id, user.id);
  }
}

// courses/courses.service.ts
@Injectable()
export class CoursesService {
  constructor(
    private readonly repository: CoursesRepository,
    private readonly enrollmentsService: EnrollmentsService,
    private readonly progressService: ProgressService,
    @Inject(CACHE_MANAGER) private cacheManager: Cache,
  ) {}

  async findAll(filters: CoursesFilterDto): Promise<Course[]> {
    const cacheKey = `courses:${JSON.stringify(filters)}`;
    
    // Check cache
    const cached = await this.cacheManager.get<Course[]>(cacheKey);
    if (cached) return cached;

    // Query database
    const courses = await this.repository.findAll(filters);

    // Cache for 5 minutes
    await this.cacheManager.set(cacheKey, courses, 300);

    return courses;
  }

  async getCourseWithProgress(
    courseId: string,
    userId: string,
  ): Promise<CourseWithProgress> {
    const course = await this.repository.findById(courseId);
    const enrollment = await this.enrollmentsService.findOrCreate(
      userId,
      courseId,
    );
    const progress = await this.progressService.calculate(enrollment.id);

    return {
      ...course,
      enrollment,
      progress,
    };
  }

  @Transactional()
  async create(dto: CreateCourseDto): Promise<Course> {
    const slug = slugify(dto.title);
    
    // Check unique slug
    const exists = await this.repository.findBySlug(slug);
    if (exists) {
      throw new ConflictException('Course with this title already exists');
    }

    const course = await this.repository.create({
      ...dto,
      slug,
      status: 'draft',
    });

    // Clear cache
    await this.cacheManager.del('courses:*');

    return course;
  }
}
```

### Guards e Decorators

```typescript
// common/guards/jwt-auth.guard.ts
@Injectable()
export class JwtAuthGuard extends AuthGuard('jwt') {
  canActivate(context: ExecutionContext) {
    return super.canActivate(context);
  }

  handleRequest(err, user, info) {
    if (err || !user) {
      throw new UnauthorizedException('Invalid token');
    }
    return user;
  }
}

// common/guards/roles.guard.ts
@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.get<string[]>(
      'roles',
      context.getHandler(),
    );
    
    if (!requiredRoles) {
      return true;
    }

    const { user } = context.switchToHttp().getRequest();
    return requiredRoles.some((role) => user.roles?.includes(role));
  }
}

// common/decorators/roles.decorator.ts
export const Roles = (...roles: string[]) => SetMetadata('roles', roles);

// Usage
@Controller('admin')
@UseGuards(JwtAuthGuard, RolesGuard)
export class AdminController {
  @Get('users')
  @Roles('ADMIN')
  async getUsers() {
    // Only ADMIN can access
  }
}
```

### Video Player Component

```typescript
// components/academy/VideoPlayer.tsx
'use client';

import { useRef, useEffect } from 'react';
import Plyr from 'plyr';

interface VideoPlayerProps {
  url: string;
  onProgress: (seconds: number) => void;
  onComplete: () => void;
  startTime?: number;
}

export function VideoPlayer({
  url,
  onProgress,
  onComplete,
  startTime = 0,
}: VideoPlayerProps) {
  const playerRef = useRef<Plyr | null>(null);
  const progressInterval = useRef<NodeJS.Timeout>();

  useEffect(() => {
    const player = new Plyr('#player', {
      controls: ['play', 'progress', 'current-time', 'mute', 'volume', 'fullscreen'],
      quality: { default: 720, options: [1080, 720, 480, 360] },
      speed: { selected: 1, options: [0.5, 0.75, 1, 1.25, 1.5, 2] },
    });

    player.currentTime = startTime;
    playerRef.current = player;

    // Track progress every 10s
    progressInterval.current = setInterval(() => {
      if (player.playing) {
        onProgress(player.currentTime);
      }
    }, 10000);

    // Track completion (90% watched)
    player.on('timeupdate', () => {
      const progress = (player.currentTime / player.duration) * 100;
      if (progress >= 90) {
        onComplete();
      }
    });

    return () => {
      clearInterval(progressInterval.current);
      player.destroy();
    };
  }, [url]);

  return (
    <div className="relative">
      <video id="player" src={url} />
    </div>
  );
}
```
